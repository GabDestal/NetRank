58def p1_is_winner(match_id: str) -> bool: 

59 """Seeded, per-match coin flip (stable whatever rows are filtered): is the winner listed as p1?""" 

60 return zlib.crc32(f"{SEED}:{match_id}".encode()) % 2 == 0

63def split_of(season: int) -> str: 

64 """train / valid / test by season.""" 

65 if season <= TRAIN_LAST: 

66 return "train" 

67 return "valid" if season <= VALID_LAST else "test"

70def oriented(pair, p1_won: bool) -> tuple[float | None, float | None]: 

71 """(odds of p1, odds of p2) from a [winnerOdds, loserOdds] pair.""" 

72 if not pair: 

73 return None, None 

74 return (pair[0], pair[1]) if p1_won else (pair[1], pair[0])

77def make_row(match: dict, values: dict) -> dict: 

78 """Features plus label and the metadata used for evaluation (never used as features).""" 

79 p1_won = p1_is_winner(match["id"]) 

80 ranks = (match["winnerRank"], match["loserRank"]) if p1_won else (match["loserRank"], match["winnerRank"]) 

81 pin, b365 = oriented(match["pinnacle"], p1_won), oriented(match["bet365"], p1_won) 

82 return { 

83 **values, 

84 "label": int(p1_won), 

85 "matchId": match["id"], 

86 "date": match["date"], 

87 "season": match["date"].year, 

88 "split": split_of(match["date"].year), 

89 "retired": bool(match["retired"]), 

90 "rankP1": ranks[0], 

91 "rankP2": ranks[1], 

92 "pinnacleP1": pin[0], 

93 "pinnacleP2": pin[1], 

94 "bet365P1": b365[0], 

95 "bet365P2": b365[1], 

96 }

99def build_dataset(db: Neo4jClient, matches: list[dict], first_season: int = FIRST_SEASON) -> pd.DataFrame: 

100 """Replay every match by date; emit a row for each played match from `first_season`.""" 

101 history = F.History() 

102 snapshots: dict[date, F.Snapshot] = {} 

103 rows = [] 

104 for day, group in groupby(matches, key=lambda m: m["date"]): 

105 played = [m for m in group if not m["walkover"]] 

106 if day.year >= first_season: 106 ↛ 115line 106 didn't jump to line 115 because the condition on line 106 was always true

107 cutoff = F.season_cutoff(day) 

108 if cutoff not in snapshots: 

109 log.info(f"snapshot {cutoff}") 

110 snapshots[cutoff] = F.build_snapshot(db, cutoff) 

111 for match in played: 

112 p1_won = p1_is_winner(match["id"]) 

113 players = (match["winnerId"], match["loserId"]) if p1_won else (match["loserId"], match["winnerId"]) 

114 rows.append(make_row(match, F.feature_vector(snapshots[cutoff], history, players, match["surface"]))) 

115 for match in played: # recorded only after every match of the day was scored 

116 history.record(match["winnerId"], match["loserId"], match["surface"]) 

117 return pd.DataFrame(rows)

123def logistic(c: float): 

124 """Scaled logistic regression without intercept (features are antisymmetric).""" 

125 return make_pipeline(

131def gradient_boosting(): 

132 """Gradient-boosted trees, fixed seed, no early stopping (deterministic).""" 

133 return HistGradientBoostingClassifier(random_state=SEED, learning_rate=0.05, max_iter=300, early_stopping=False)

136def scores(y, p) -> dict: 

137 """Log loss, Brier score and accuracy for probabilities p of label 1.""" 

138 p = np.clip(np.asarray(p, dtype=float), 1e-6, 1 - 1e-6) 

139 return { 

140 "logLoss": round(float(log_loss(y, p, labels=[0, 1])), 4), 

141 "brier": round(float(brier_score_loss(y, p)), 4), 

142 "accuracy": round(float(accuracy_score(y, p > 0.5)), 4), 

143 "n": int(len(y)), 

144 }

152def select_model(train: pd.DataFrame, valid: pd.DataFrame) -> tuple[dict, list[dict]]: 

153 """Validation-only model selection; returns the chosen spec and every candidate's scores.""" 

154 x_train, y_train, x_valid, y_valid = matrix(train), train["label"], matrix(valid), valid["label"] 

155 candidates = [] 

156 for c in C_GRID: 

157 model = logistic(c).fit(x_train, y_train) 

158 candidates.append({"model": "logistic", "C": c, **scores(y_valid, model.predict_proba(x_valid)[:, 1])}) 

159 best_lr = min(candidates, key=lambda r: r["logLoss"]) 

160 gbm = gradient_boosting().fit(x_train, y_train) 

161 candidates.append({"model": "gradient_boosting", "C": None, **scores(y_valid, gbm.predict_proba(x_valid)[:, 1])}) 

162 gbm_wins = candidates[-1]["logLoss"] < best_lr["logLoss"] * (1 - GBM_MARGIN) 

163 chosen = {"model": "gradient_boosting"} if gbm_wins else {"model": "logistic", "C": best_lr["C"]} 

164 return chosen, candidates

167def rank_feature(frame: pd.DataFrame) -> np.ndarray: 

168 """log(rank p2) - log(rank p1): positive when p1 is better ranked; 0 when a rank is missing.""" 

169 r1, r2 = pd.to_numeric(frame["rankP1"]), pd.to_numeric(frame["rankP2"]) 

170 value = np.log(r2) - np.log(r1) 

171 return value.fillna(0.0).to_numpy().reshape(-1, 1)

174def fit_rank_baseline(train: pd.DataFrame): 

175 """One-feature logistic model on the rank ratio, fitted on training rows that have both ranks.""" 

176 known = train["rankP1"].notna() & train["rankP2"].notna() 

177 return LogisticRegression(fit_intercept=False).fit(rank_feature(train[known]), train.loc[known, "label"])

183def bookmaker_probability(frame: pd.DataFrame, book: str) -> np.ndarray: 

184 """Margin-free probability that p1 wins from one bookmaker's odds.""" 

185 return np.array([F.devig(o1, o2) for o1, o2 in zip(frame[f"{book}P1"], frame[f"{book}P2"])])

190 return {

198def compare(frame: pd.DataFrame, model, rank_model) -> dict: 

199 """Scores of every predictor on the same rows.""" 

200 return {name: scores(frame["label"], p) for name, p in predictors(frame, model, rank_model).items()}

203def final_test(dataset: pd.DataFrame, chosen: dict) -> dict: 

204 """Refit on train + validation, then the single evaluation on the test seasons.""" 

205 fit_rows = dataset[dataset["split"] != "test"] 

206 model = (logistic(chosen["C"]) if chosen["model"] == "logistic" else gradient_boosting()).fit( 

207 matrix(fit_rows), fit_rows["label"] 

208 ) 

209 rank_model = fit_rank_baseline(dataset[dataset["split"] == "train"]) 

210 test = dataset[dataset["split"] == "test"] 

211 priced = test["pinnacleP1"].notna() & test["bet365P1"].notna() 

212 common, retired = test[priced & ~test["retired"]], test[priced & test["retired"]] 

213 completed = test[~test["retired"]] 

214 return { 

215 "model": model, 

216 "common": compare(common, model, rank_model), 

217 "retired": compare(retired, model, rank_model), 

218 "allCompleted": scores(completed["label"], model.predict_proba(matrix(completed))[:, 1]), 

219 "calibration": {"labels": common["label"].to_numpy(), "probabilities": predictors(common, model, rank_model)}, 

220 "seasons": (int(test["season"].min()), int(test["season"].max())), 

221 "lastDate": str(test["date"].max()), 

222 }

228def export(model, chosen: dict, serving: F.Snapshot, summary: dict) -> dict: 

229 """Write the JSON artifact read by the API (logistic regression only).""" 

230 if chosen["model"] != "logistic": 

231 raise RuntimeError("Only logistic regression can be served from JSON; add a serving path first.") 

232 scaler, regression = model.named_steps["standardscaler"], model.named_steps["logisticregression"] 

233 weights = (regression.coef_[0] / scaler.scale_).round(10).tolist() 

234 digest = hashlib.sha256(json.dumps([list(F.FEATURES), weights]).encode()).hexdigest()[:8] 

235 created = datetime.now(timezone.utc) 

236 artifact = { 

237 "version": f"logistic-{created:%Y%m%d}-{digest}", 

238 "createdAt": created.isoformat(timespec="seconds"), 

239 "kind": "logistic", 

240 "features": list(F.FEATURES), 

241 "weights": weights, 

242 "standardisedWeights": dict(zip(F.FEATURES, regression.coef_[0].round(6).tolist())), 

243 "trainedOn": {"firstSeason": FIRST_SEASON, "lastSeason": VALID_LAST, "C": chosen["C"], "seed": SEED}, 

244 "metrics": summary, 

245 "servingSnapshot": { 

246 "cutoff": serving.cutoff.isoformat(), 

247 "prestige": serving.prestige, 

248 "recent": serving.recent, 

249 "embeddings": {k: [round(x, 6) for x in v] for k, v in serving.embeddings.items()}, 

250 }, 

251 } 

252 MODEL_PATH.parent.mkdir(parents=True, exist_ok=True) 

253 MODEL_PATH.write_text(json.dumps(artifact), encoding="utf-8") 

254 return artifact

257def main() -> None: 

258 """Build, select, test once, export and report.""" 

259 logging.basicConfig(level=logging.INFO, format="%(message)s") 

260 db = Neo4jClient.from_settings(get_settings()) 

261 try: 

262 matches = db.read(MATCHES) 

263 dataset = build_dataset(db, matches) 

264 train, valid = dataset[dataset["split"] == "train"], dataset[dataset["split"] == "valid"] 

265 log.info(f"rows: {dataset['split'].value_counts().to_dict()}; p1 won rate {dataset['label'].mean():.4f}") 

266 chosen, candidates = select_model(train, valid) 

267 log.info(f"validation: chosen {chosen}") 

268 reference = fit_rank_baseline(train).predict_proba(rank_feature(valid))[:, 1] 

269 reference_scores = scores(valid["label"], reference) 

270 candidates.append({"model": "ranking baseline (reference only)", "C": None, **reference_scores}) 

271 result = final_test(dataset, chosen) 

272 serving_cutoff = max(m["date"] for m in matches) + timedelta(days=1) 

273 serving = F.build_snapshot(db, serving_cutoff) 

274 summary = {"test": result["common"], "validation": candidates} 

275 artifact = export(result["model"], chosen, serving, summary) 

276 report_model.write(dataset, chosen, candidates, result, artifact) 

277 log.info(f"model {artifact['version']} written to {MODEL_PATH}") 

278 finally: 

279 db.close()
