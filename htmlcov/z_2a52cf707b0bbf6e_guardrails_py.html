1"""Guardrails for LLM-generated Cypher and for the agent's scope (decision log D62-D63). 

2 

3Cypher passes five independent layers; any one is enough to stop a write: 

41. static text checks: one statement, no write clauses, no LOAD CSV, procedures only from an allow-list; 

52. schema check: every label and relationship type must exist in the NetRank schema; 

63. EXPLAIN in a READ transaction: syntax errors, write operators in the plan and the server's 

7 unknown-label / unknown-type notifications are rejected; 

84. LIMIT enforced (appended when missing, capped when too large); 

95. execution in a READ-mode transaction (the server refuses writes) with a timeout and a row cap. 

10"""

19MAX_LIMIT = 50 

20TIMEOUT_SECONDS = 5.0 

21LABELS = frozenset({"Player", "Match", "Tournament", "Bookmaker"}) 

22RELATIONSHIP_TYPES = frozenset({"WON", "LOST", "PART_OF", "PRICED", "BEAT", "SIMILAR_STYLE"}) 

23ALLOWED_PROCEDURES = frozenset({"db.index.fulltext.querynodes", "db.labels", "db.relationshiptypes"})

39WRITE_CLAUSES = re.compile( 

40 r"""\b( 

41 CREATE | MERGE | DELETE | DETACH | SET | REMOVE | DROP | FOREACH 

42 | LOAD\s+CSV | PERIODIC\s+COMMIT | IN\s+TRANSACTIONS 

43 | GRANT | DENY | REVOKE | ALTER | RENAME | START\s+DATABASE | STOP\s+DATABASE 

44 )\b""", 

45 re.IGNORECASE | re.VERBOSE, 

46) 

47# CALL followed by a dotted name is a procedure; CALL { or CALL ( is a subquery. 

48PROCEDURE_CALL = re.compile(r"\bCALL\s+([A-Za-z_][\w.]*)\s*\(", re.IGNORECASE) 

49NODE_LABELS = re.compile(r"\(\s*\w*\s*((?::\s*`?\w+`?\s*)+)") 

50REL_TYPES = re.compile(r"\[\s*\w*\s*:\s*([`\w|:\s]+?)\s*(?:\*|\{|\]|WHERE)", re.IGNORECASE) 

51TRAILING_LIMIT = re.compile(r"\bLIMIT\s+(\d+)\s*$", re.IGNORECASE) 

52STRING_LITERAL = re.compile(r"'(?:[^'\\]|\\.)*'|\"(?:[^\"\\]|\\.)*\"") 

53COMMENT = re.compile(r"//[^\n]*|/\*.*?\*/", re.DOTALL) 

54# Procedure calls are covered by the allow-list (layer 1) and the READ transaction (layer 5). 

55WRITE_OPERATORS = ("Create", "Merge", "Delete", "Set", "Remove", "Foreach", "LoadCSV") 

56# GQL status codes: label / relationship type / property key does not exist. 

57SCHEMA_NOTIFICATIONS = {"01N50": "unknown label", "01N51": "unknown relationship type", "01N52": "unknown property key"}

60class GuardrailError(ValueError): 

61 """Generated Cypher was rejected; `layer` names the check that stopped it.""" 

62 

63 def __init__(self, layer: str, message: str) -> None: 

64 """Keep the layer for the trace and the evaluation report.""" 

65 super().__init__(f"[{layer}] {message}") 

66 self.layer = layer

69@dataclass(frozen=True) 

70class CheckedQuery: 

71 """A query that passed every static and EXPLAIN check, with the LIMIT actually applied.""" 

72 

73 text: str 

74 limit: int 

75 

76 

77def strip_literals_and_comments(query: str) -> str: 

78 """Query text with strings and comments blanked, so keywords inside them are not matched.""" 

79 return STRING_LITERAL.sub("''", COMMENT.sub(" ", query)) 

80 

81 

82def static_check(query: str) -> str: 

83 """Layer 1: single statement, read-only clauses, allowed procedures. Returns the trimmed query.""" 

84 text = query.strip().rstrip(";").strip() 

85 if not text: 

86 raise GuardrailError("static", "empty query") 

87 code = strip_literals_and_comments(text) 

88 if ";" in code: 

89 raise GuardrailError("static", "only one statement is allowed") 

90 if match := WRITE_CLAUSES.search(code): 

91 raise GuardrailError("static", f"write or admin clause not allowed: {match.group(1).upper()}") 

92 for name in PROCEDURE_CALL.findall(code): 

93 if name.lower() not in ALLOWED_PROCEDURES: 

94 raise GuardrailError("static", f"procedure not allowed: {name}") 

95 return text 

96 

97 

98def schema_check(query: str) -> None: 

99 """Layer 2: labels and relationship types must belong to the NetRank schema.""" 

100 code = strip_literals_and_comments(query) 

101 labels = {name.strip(" `") for group in NODE_LABELS.findall(code) for name in group.split(":")} - {""} 

102 types = {name.strip(" `:") for group in REL_TYPES.findall(code) for name in group.split("|")} - {""} 

103 if unknown := sorted(labels - LABELS): 

104 raise GuardrailError("schema", f"unknown label(s) {unknown}; labels are {sorted(LABELS)}") 

105 if unknown := sorted(types - RELATIONSHIP_TYPES): 

106 known = sorted(RELATIONSHIP_TYPES) 

107 raise GuardrailError("schema", f"unknown relationship type(s) {unknown}; types are {known}") 

108 

109 

110def enforce_limit(query: str) -> CheckedQuery: 

111 """Layer 4: a trailing LIMIT no larger than MAX_LIMIT (appended when missing, lowered when larger).""" 

112 if match := TRAILING_LIMIT.search(query): 

113 limit = min(int(match.group(1)), MAX_LIMIT) 

114 return CheckedQuery(text=query[: match.start()] + f"LIMIT {limit}", limit=limit) 

115 return CheckedQuery(text=f"{query}\nLIMIT {MAX_LIMIT}", limit=MAX_LIMIT) 

116 

117 

118def operators(plan: dict) -> list[str]: 

119 """Operator names of a plan tree (planner suffix removed).""" 

120 return [plan["operatorType"].split("@")[0]] + [op for child in plan.get("children", []) for op in operators(child)] 

121 

122 

123def explain_check(db: Neo4jClient, query: str) -> None: 

124 """Layer 3: the server parses and plans the query in a READ transaction without running it.""" 

125 try: 

126 plan, notifications = db.explain(query, {}) 

127 except CypherSyntaxError as exc: 

128 raise GuardrailError("explain", f"syntax error: {exc.message}") from exc 

129 except ClientError as exc: 

130 raise GuardrailError("explain", f"rejected by the server: {exc.message}") from exc 

131 if writes := [op for op in operators(plan) if op.startswith(WRITE_OPERATORS)]: 

132 raise GuardrailError("explain", f"plan contains write operators {writes}") 

133 if unknown := [description for code, description in notifications if code in SCHEMA_NOTIFICATIONS]: 

134 message = "server reports: " + "; ".join(sorted({u.rstrip(". ") for u in unknown})) 

135 if any(code == "01N52" for code, _ in notifications): 135 ↛ 137line 135 didn't jump to line 137 because the condition on line 135 was always true

136 message += ". Known properties: " + property_hint(query) 

137 raise GuardrailError("explain", message) 

138 

139 

140def property_hint(query: str) -> str: 

141 """Property keys of the labels and relationship types the query mentions (all of them if none).""" 

142 code = strip_literals_and_comments(query) 

143 used = [name for name in PROPERTY_KEYS if re.search(r":\s*`?" + name + r"(?!\w)", code)] or list(PROPERTY_KEYS) 

144 return "; ".join(f"{name}: {PROPERTY_KEYS[name]}" for name in used) 

145 

146 

147def check(db: Neo4jClient, query: str) -> CheckedQuery: 

148 """Layers 1-4; returns the query that may be executed.""" 

149 text = static_check(query) 

150 schema_check(text) 

151 checked = enforce_limit(text) 

152 explain_check(db, checked.text) 

153 return checked 

154 

155 

156def run_readonly(db: Neo4jClient, query: str) -> dict: 

157 """Check, then execute in a READ-mode transaction with timeout and row cap (layer 5).""" 

158 checked = check(db, query) 

159 try: 

160 rows = db.read_only(checked.text, {}, timeout=TIMEOUT_SECONDS, max_rows=checked.limit) 

161 except ClientError as exc: 

162 raise GuardrailError("execute", f"rejected at execution: {exc.message}") from exc 

163 return {"cypher": checked.text, "rows": rows, "rowCount": len(rows), "limit": checked.limit}
