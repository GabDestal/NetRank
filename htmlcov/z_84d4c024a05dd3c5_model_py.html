1"""The served winner model: a logistic regression exported to JSON (no scikit-learn in the API image).

3Artifact (written by scripts/train.py, read here): 

4 version, createdAt, kind ("logistic"), features [names], weights [w per feature, already divided by

6P(p1 wins) = sigmoid(sum_j w_j * x_j). No intercept: features are antisymmetric, so

30 def __init__(self, artifact: dict) -> None:

45 def logit(self, features: dict[str, float]) -> float: 

46 """Linear score; positive favours p1.""" 

47 return sum(self.weights[name] * features[name] for name in FEATURES) 

48 

49 def probability(self, features: dict[str, float]) -> float: 

50 """P(p1 wins).""" 

51 return 1.0 / (1.0 + math.exp(-self.logit(features))) 

52 

53 def contributions(self, features: dict[str, float], top: int = 5) -> list[dict]: 

54 """Largest |w * x| terms, with the 64 FastRP terms summed into one 'fastrpEmbedding' entry.""" 

55 terms = {name: self.weights[name] * features[name] for name in FEATURES if not name.startswith("fastrp")} 

56 terms["fastrpEmbedding"] = sum(self.weights[n] * features[n] for n in FEATURES if n.startswith("fastrp")) 

57 ranked = sorted(terms.items(), key=lambda item: abs(item[1]), reverse=True)[:top] 

58 return [{"feature": name, "logitContribution": round(value, 4)} for name, value in ranked]
