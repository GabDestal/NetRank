147def compact_schema(node: Any) -> Any: 

148 """JSON schema without what the model does not need: titles, model docstrings, null defaults. 

149 

150 `anyOf: [X, null]` becomes X with a nullable type. The tool specs are sent on every turn, so 

151 this is most of the prompt-size cut of decision log D72. 

152 """ 

153 if isinstance(node, list): 

154 return [compact_schema(item) for item in node] 

155 if not isinstance(node, dict): 

156 return node 

157 out = { 

158 key: compact_schema(value) 

159 for key, value in node.items() 

160 if key != "title" 

161 and not (key == "description" and "properties" in node) 

162 and not (key == "default" and value is None) 

163 } 

164 options = out.get("anyOf") 

165 if isinstance(options, list) and len(options) == 2 and {"type": "null"} in options: 

166 kept = dict(next(option for option in options if option != {"type": "null"}))

171 out = {**{k: v for k, v in out.items() if k != "anyOf"}, **kept}

215def trim_for_llm(value: Any, omit: frozenset[str] = frozenset(), limit: int = LLM_LIST_LIMIT, depth: int = 0) -> Any: 

216 """What the LLM sees of a result; the trace (and so the verifier) keeps the full result. 

217 

218 Lists are cut to `limit` items, null values dropped, floats rounded to 4 decimals (the verifier 

219 rounds the full value to the precision the model writes, so cited numbers still match), player 

220 biography fields dropped below the top level, and the tool's `omit` keys left out (decision log D72). 

221 """ 

222 if isinstance(value, dict): 

223 hidden = omit | (BIOGRAPHY if depth else frozenset()) 

224 return { 

225 key: trim_for_llm(item, omit, limit, depth + 1) 

226 for key, item in value.items() 

227 if key not in hidden and item is not None 

228 } 

229 if isinstance(value, list): 

230 trimmed = [trim_for_llm(item, omit, limit, depth + 1) for item in value[:limit]] 

231 return trimmed + ([f"... {len(value) - limit} more"] if len(value) > limit else []) 

232 if isinstance(value, float): 

233 return round(value, 4) 

234 return value

293 def search_player(self, args: NameArgs) -> dict: 

294 """Unique when exactly one hit or an exact full-name match; otherwise ambiguous (never guessed). 

295 

296 The name must use only words from the question or from earlier results: in live runs the model 

297 expanded "Zverev" to "Alexander Zverev" by itself, which is a guess (decision log D61). 

298 """ 

299 if self.question_words: 299 ↛ 306line 299 didn't jump to line 306 because the condition on line 299 was always true

300 extra = sorted(set(re.findall(r"[a-z0-9]+", fold(args.name))) - self.known_words()) 

301 if extra: 

302 raise InvalidRequestError( 

303 f"search with the name exactly as the user wrote it; {extra} is not in the question. " 

304 + "If that search is ambiguous, ask the user which player they mean"

306 page = players.search(self._db, args.name, 1, 5) 

307 candidates = [{"id": h.id, "name": h.name, "ioc": h.ioc} for h in page.items]

336 def call(self, name: str, raw_args: dict) -> tuple[ToolCallRecord, str]: 

337 """Run one tool call; returns its trace record and the JSON content shown to the LLM.""" 

338 call_id = f"t{len(self.trace) + 1}" 

339 record = ToolCallRecord(id=call_id, tool=name, arguments=raw_args, success=False) 

340 try: 

341 if name not in self.handlers: 

342 raise InvalidRequestError(f"unknown tool {name}") 

343 args = TOOL_ARGS[name].model_validate(raw_args) 

344 result = self.handlers[name](args) 

345 data = result.model_dump(mode="json") if isinstance(result, BaseModel) else result 

346 record.success, record.result = True, data 

347 self.resolved |= ids_in(data) - (self.ambiguous if name == "search_player" else set()) 

348 shown = trim_for_llm(data, LLM_OMIT.get(name, frozenset()), LLM_LIMITS.get(name, LLM_LIST_LIMIT)) 

349 content = {"toolCallId": call_id, "result": shown} 

350 except guardrails.GuardrailError as exc: 

351 record.error, record.guardrail = str(exc), exc.layer 

352 content = {"toolCallId": call_id, "error": f"Guardrail rejected the query: {exc}"} 

353 except (ValidationError, InvalidRequestError, NotFoundError, GdsUnavailableError) as exc: 

354 record.error = str(exc).splitlines()[0][:300] 

355 content = {"toolCallId": call_id, "error": record.error} 

356 self.trace.append(record) 

357 return record, json.dumps(content, default=str)
