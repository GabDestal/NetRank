32def table(rows: dict[str, dict], first: str = "Predictor") -> str: 

33 """Markdown table from {name: {metric: value}}.""" 

34 metrics = ["logLoss", "brier", "accuracy", "n"] 

35 lines = [f"| {first} | Log loss | Brier | Accuracy | Rows |", "| --- | ---: | ---: | ---: | ---: |"] 

36 for name, values in rows.items(): 

37 lines.append(f"| {name} | " + " | ".join(str(values[m]) for m in metrics) + " |") 

38 return "\n".join(lines)

41def calibration_bins(labels, probabilities: dict) -> dict[str, tuple]: 

42 """(mean predicted, observed frequency) per quantile bin for each predictor.""" 

43 curves = {} 

44 for name, p in probabilities.items(): 

45 observed, predicted = calibration_curve(labels, p, n_bins=BINS, strategy="quantile") 

46 curves[name] = (predicted, observed) 

47 return curves

50def plot_calibration(curves: dict[str, tuple], rows: int) -> None: 

51 """Reliability diagram: one line per predictor against the diagonal of perfect calibration.""" 

52 FIGURE.parent.mkdir(parents=True, exist_ok=True) 

53 fig, ax = plt.subplots(figsize=(7.2, 6.0), dpi=150, facecolor=SURFACE)

56 ax.text(0.03, 0.08, "perfect calibration", color=MUTED, fontsize=8, ha="left", rotation=40)

58 color, marker = SERIES[name] 

59 ax.plot( 

60 predicted, 

61 observed, 

62 color=color,

91def calibration_table(curves: dict[str, tuple]) -> str: 

92 """Text version of the figure (bin means per predictor).""" 

93 lines = ["| Predictor | Predicted (bin mean) -> observed |", "| --- | --- |"] 

94 for name, (predicted, observed) in curves.items(): 

95 pairs = ", ".join(f"{p:.2f}->{o:.2f}" for p, o in zip(predicted, observed)) 

96 lines.append(f"| {name} | {pairs} |") 

97 return "\n".join(lines)

100def coefficient_table(artifact: dict) -> str: 

101 """Standardised weights of the scalar features, largest first (FastRP summarised).""" 

102 weights = artifact["standardisedWeights"] 

103 scalar = sorted(((n, weights[n]) for n in F.SCALAR_FEATURES), key=lambda item: abs(item[1]), reverse=True) 

104 fastrp = [abs(weights[n]) for n in F.EMBEDDING_FEATURES] 

105 lines = ["| Feature (p1 - p2) | Weight per standard deviation |", "| --- | ---: |"] 

106 lines += [f"| {name} | {value:+.3f} |" for name, value in scalar] 

107 lines.append(f"| fastrp00..63 (64 dims) | mean abs {np.mean(fastrp):.3f}, max abs {max(fastrp):.3f} |") 

108 return "\n".join(lines)

111def reading_notes(dataset: pd.DataFrame, result: dict) -> str: 

112 """Facts needed to read the tables: gaps to the market and the collinear PageRank pair.""" 

113 train = dataset[dataset["split"] == "train"] 

114 correlation = train["logPrestige"].corr(train["logRecentPrestige"]) 

115 common = result["common"] 

116 gap_market = common["Graph model"]["logLoss"] - common["Pinnacle (no margin)"]["logLoss"] 

117 gap_rank = common["Graph model"]["logLoss"] - common["Ranking baseline"]["logLoss"] 

118 return "\n".join( 

119 [ 

120 f"- Log loss gap to Pinnacle: {gap_market:+.4f}; to the ranking baseline: {gap_rank:+.4f}" 

121 + " (positive = worse).", 

122 f"- logPrestige and logRecentPrestige are correlated (r = {correlation:.2f} on train), so their weights" 

123 + " are not interpretable one by one: together they reward recent prestige and penalise a career" 

124 + " prestige that recent results no longer support.", 

125 "- The ranking baseline uses ATP ranks, which summarise 52 weeks of results at every level" 

126 + " (Challengers, qualifying) that are not in this graph; the graph model sees tour-level matches only.", 

127 ] 

128 )

131def write(dataset: pd.DataFrame, chosen: dict, candidates: list[dict], result: dict, artifact: dict) -> None: 

132 """Assemble the Markdown report and the figure.""" 

133 curves = calibration_bins(result["calibration"]["labels"], result["calibration"]["probabilities"]) 

134 plot_calibration(curves, len(result["calibration"]["labels"])) 

135 splits = dataset.groupby("split").agg( 

136 rows=("label", "size"), p1WonRate=("label", "mean"), first=("date", "min"), last=("date", "max") 

137 ) 

138 validation = {(f"{c['model']} C={c['C']}" if c["C"] is not None else c["model"]): c for c in candidates} 

139 first, last = result["seasons"] 

140 lines = [ 

141 "# Match-outcome model report", 

142 "", 

143 f"Generated by `make train` (`scripts/train.py`). Model `{artifact['version']}`.", 

144 "", 

145 "## Data and split", 

146 "", 

147 "One row per played match (walkovers excluded) from 1995; p1/p2 assigned by a seeded coin flip per " 

148 + "match; label = p1 won. Features are p1 - p2 differences computed only from matches before the " 

149 + "match (graph snapshot at Jan 1 of its season; history strictly before its date).", 

150 "", 

151 "| Split | Rows | p1 won rate | First match | Last match |", 

152 "| --- | ---: | ---: | --- | --- |", 

153 *[f"| {s} | {r.rows:,} | {r.p1WonRate:.4f} | {r.first} | {r.last} |" for s, r in splits.iterrows()], 

154 "", 

155 "## Model selection (validation 2019-2021 only)", 

156 "", 

157 table(validation, "Candidate"), 

158 "", 

159 f"Chosen: **{chosen['model']}**"

162 + "The chosen model is refitted on train + validation before the test.",

168 + "The ranking baseline is a one-feature logistic model on log(rank p2 / rank p1), fitted on train.",

172 "Retired matches (same predictors, reported separately):", 

173 "",
