11# Full-text index lookup (Lucene). Wildcard terms all score the same, so ties are broken by career 

12# length (a degree count read from the node, no traversal): 'feder' ranks Roger Federer before 

13# Federico Browne. Hits are few, so they are collected once and sliced for the page; the cost 

14# grows with the number of hits, not with the size of the graph. 

15SEARCH = """ 

16CALL db.index.fulltext.queryNodes('player_name_ft', $query) YIELD node, score 

17WITH node, score, COUNT { (node)-[:WON|LOST]->() } AS matches 

18ORDER BY score DESC, matches DESC, node.name 

19WITH collect(node {.id, .name, .ioc, .hand, .dob, score: score}) AS hits 

20RETURN size(hits) AS total, hits[$skip..$skip + $limit] AS items 

21"""

23# One unique-index seek on Player.id, then two subqueries over that player's matches only 

24# (bounded by career length, ~1,500 at most). Conditional counting in one pass avoids an 

25# OPTIONAL MATCH per measure, which would multiply rows. 

26PROFILE = """ 

27MATCH (p:Player {id: $id}) 

28CALL (p) { 

29 MATCH (p)-[r:WON|LOST]->(m:Match) 

30 RETURN count(m) AS matches, 

31 count(CASE WHEN type(r) = 'WON' AND NOT m.walkover THEN 1 END) AS wins, 

32 count(CASE WHEN type(r) = 'LOST' AND NOT m.walkover THEN 1 END) AS losses, 

33 count(CASE WHEN type(r) = 'WON' AND m.round = 'F' THEN 1 END) AS titles, 

34 min(r.rank) AS bestRank, min(m.date) AS firstMatch, max(m.date) AS lastMatch 

35} 

36CALL (p) { 

37 MATCH (p)-[r:WON|LOST]->(m:Match) 

38 WHERE NOT m.walkover AND m.surface IS NOT NULL 

39 WITH m.surface AS surface, 

40 count(CASE WHEN type(r) = 'WON' THEN 1 END) AS wins, 

41 count(CASE WHEN type(r) = 'LOST' THEN 1 END) AS losses 

42 ORDER BY wins + losses DESC 

43 RETURN collect({surface: surface, wins: wins, losses: losses}) AS bySurface 

44} 

45RETURN p {.id, .name, .ioc, .hand, .dob, .height, .wikidataId} AS player, 

46 matches, wins, losses, titles, bestRank, firstMatch, lastMatch, bySurface 

47"""

49# Anchored on the player, so filters apply to that player's matches only (no global date scan). 

50# The page is cut before the per-match lookups (tournament, opponent, odds), which therefore run 

51# for at most $limit matches whatever the career length. 

52MATCHES = """ 

53MATCH (p:Player {id: $id})-[r:WON|LOST]->(m:Match) 

54WHERE ($surface IS NULL OR m.surface = $surface) 

55 AND ($since IS NULL OR m.date >= $since) 

56 AND ($until IS NULL OR m.date <= $until) 

57WITH p, r, m 

58ORDER BY m.date DESC, m.id DESC 

59WITH p, collect({won: type(r) = 'WON', m: m}) AS rows 

60CALL (p, rows) { 

61 UNWIND rows[$skip..$skip + $limit] AS row 

62 WITH p, row.won AS won, row.m AS m 

63 MATCH (m)-[:PART_OF]->(t:Tournament) 

64 MATCH (m)<-[:WON|LOST]-(o:Player) 

65 WHERE o <> p 

66 RETURN collect({ 

67 matchId: m.id, date: m.date, tournament: t.name, tournamentId: t.id, level: t.levelName, 

68 surface: m.surface, round: m.round, result: CASE WHEN won THEN 'W' ELSE 'L' END, 

69 score: m.score, walkover: m.walkover, retired: m.retired, 

70 opponent: o {.id, .name, .ioc, .hand, .dob}, 

71 odds: [(b:Bookmaker)-[pr:PRICED]->(m) | { 

72 bookmaker: b.name, 

73 playerOdds: CASE WHEN won THEN pr.winnerOdds ELSE pr.loserOdds END, 

74 opponentOdds: CASE WHEN won THEN pr.loserOdds ELSE pr.winnerOdds END 

75 }] 

76 }) AS items 

77} 

78RETURN size(rows) AS total, items 

79"""

84def lucene_query(text: str) -> str: 

85 """Turn free text into a Lucene query: every word as a prefix ('rog fed' -> 'rog* AND fed*'). 

86 

87 Words are folded first because Lucene does not analyse wildcard terms. Only letters and digits 

88 survive, so user input cannot inject Lucene syntax. 

89 """ 

90 words = re.findall(r"[a-z0-9]+", fold(text)) 

91 return " AND ".join(f"{word}*" for word in words)

94def search(db: Neo4jClient, text: str, page: int, size: int) -> Page[PlayerHit]: 

95 """Players whose name matches every word of `text` as a prefix, best match first.""" 

96 query = lucene_query(text) 

97 if not query: 

98 return Page[PlayerHit](items=[], page=page, size=size, total=0) 

99 row = db.read(SEARCH, {"query": query, "skip": (page - 1) * size, "limit": size})[0] 

100 return Page[PlayerHit](items=row["items"], page=page, size=size, total=row["total"])

103def profile(db: Neo4jClient, player_id: int) -> PlayerProfile: 

104 """Biography and career aggregates; raises NotFoundError for an unknown id.""" 

105 rows = db.read(PROFILE, {"id": player_id}) 

106 if not rows: 

107 raise NotFoundError(f"No player with id {player_id}") 

108 row = rows[0] 

109 by_surface = [SurfaceRecord(**s, winRate=round(s["wins"] / (s["wins"] + s["losses"]), 4)) for s in row["bySurface"]] 

110 return PlayerProfile( 

111 **row["player"], 

112 matches=row["matches"], 

113 wins=row["wins"], 

114 losses=row["losses"], 

115 titles=row["titles"], 

116 bestRank=row["bestRank"], 

117 firstMatch=row["firstMatch"], 

118 lastMatch=row["lastMatch"], 

119 bySurface=by_surface, 

120 )

123def ensure_player(db: Neo4jClient, player_id: int) -> None: 

124 """Raise NotFoundError unless the player exists (one unique-index seek).""" 

125 if not db.read(PLAYER_EXISTS, {"id": player_id}): 

126 raise NotFoundError(f"No player with id {player_id}")

129def matches(db: Neo4jClient, player_id: int, filters: dict, page: int, size: int) -> Page[PlayerMatch]: 

130 """A player's matches, most recent first. `filters` may hold surface, since and until.""" 

131 params = { 

132 "id": player_id, 

133 "surface": filters.get("surface"), 

134 "since": filters.get("since"), 

135 "until": filters.get("until"), 

136 "skip": (page - 1) * size, 

137 "limit": size, 

138 } 

139 rows = db.read(MATCHES, params) 

140 if not rows: 

141 ensure_player(db, player_id) 

142 return Page[PlayerMatch](items=[], page=page, size=size, total=0) 

143 return Page[PlayerMatch](items=rows[0]["items"], page=page, size=size, total=rows[0]["total"])
